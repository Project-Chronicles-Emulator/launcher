cmake_minimum_required(VERSION 3.25)
project(ProjectChronicles VERSION 0.2.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
    message(FATAL_ERROR "This desktop application requires Windows.")
endif()
enable_language(RC)
include(CTest)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Boost CONFIG REQUIRED COMPONENTS asio)
find_package(unofficial-webview2 CONFIG REQUIRED)
if(BUILD_TESTING)
    find_package(GTest CONFIG REQUIRED)
endif()
include(cmake/ChroniclesResources.cmake)
include(cmake/ChroniclesPrecompiledHeaders.cmake)
include(cmake/ChroniclesModules.cmake)
add_subdirectory(src/distribution)
add_subdirectory(src/services/portal)
add_subdirectory(launcher)
